STEALTH_JS = """

def load_cookies(path):
    f = path / "cookies.json"
    if not f.exists(): return None
    return json.loads(f.read_text())

def get_proxy(index):
    p = PROXIES[index % len(PROXIES)]
    host, port, user, pwd = p.split(":")
    return {"server": f"http://{host}:{port}", "username": user, "password": pwd}

async def dismiss_popups(page):
    for txt in ["Not Now", "Decline optional cookies", "Allow essential and optional cookies", "Cancel"]:
        try:
            btn = page.get_by_role("button", name=txt)
            if await btn.is_visible(timeout=1000):
                await btn.click()
                await asyncio.sleep(0.8)
        except Exception:
            pass

async def do_follow(page, name):
    try:
        await page.goto(PROFILE_URL, wait_until="domcontentloaded", timeout=30000)
        await asyncio.sleep(3)
        await dismiss_popups(page)
        follow_btn = page.get_by_role("button", name="Follow").first
        if await follow_btn.is_visible(timeout=3000):
            await follow_btn.click()
            await asyncio.sleep(2)
            print(f"  [+] followed Unreal Sports")
        else:
            print(f"  [=] already following")
    except Exception as e:
        print(f"  follow error: {e}")

async def do_comment(page, comment, name):

async def run_account(pw, account_path, index):
    name = account_path.name
    print(f"\n[→] {name}")
    cookies = load_cookies(account_path)
    if not cookies:
        print(f"  [skip] no cookies.json")
        return name, 0, 0

    proxy = get_proxy(index)
    print(f"  proxy: {proxy['server']}")

    browser = await pw.chromium.launch(
        headless=False,
        proxy=proxy,
        args=[
            "--no-sandbox",
            "--disable-blink-features=AutomationControlled",
            "--disable-dev-shm-usage",
            "--autoplay-policy=no-user-gesture-required",
        ]
    )

    ctx = await browser.new_context(
        viewport={"width": 390, "height": 844},
        user_agent=UA,
        locale="en-US",
        timezone_id="America/New_York",
        has_touch=True,
        is_mobile=True,
    )

    await ctx.add_cookies(cookies)
    page = await ctx.new_page()
    await page.add_init_script(STEALTH_JS)

    # follow first
    await do_follow(page, name)
    await asyncio.sleep(random.uniform(2, 4))

    # post one unique comment
    comment = random.choice(COMMENTS)
    await do_comment(page, comment, name)
    await asyncio.sleep(random.uniform(2, 4))

    viewed = liked = 0

async def main():
    accounts = sorted(ACCOUNTS_DIR.iterdir()) if ACCOUNTS_DIR.exists() else []
    accounts = [a for a in accounts if a.is_dir() and (a/"cookies.json").exists()]
    if not accounts:
        print("[!] no accounts found"); return
    print(f"[+] {len(accounts)} accounts")
    print(f"[+] target: {TARGET_URL}\n")
    async with async_playwright() as pw:
        tasks = [run_account(pw, acc, i) for i, acc in enumerate(accounts)]
        results = await asyncio.gather(*tasks)
    total_v = sum(r[1] for r in results)
    total_l = sum(r[2] for r in results)
    print(f"\n  TOTAL: views={total_v} likes={total_l}")

if __name__ == "__main__":
    asyncio.run(main())
